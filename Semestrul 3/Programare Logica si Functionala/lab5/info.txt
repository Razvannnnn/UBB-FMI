CAR -> extrage primul element dintr-o lista
    (car '(rose violet daisy buttercup)) -> rose

CDR -> returneaza restul listei in afara de primul
    (cdr '(rose violet daisy buttercup)) -> (violet daisy buttercup)

CONS ->  da append la o lista
    (cons 'pine '(fir oak maple)) -> (pine fir oak maple)

    (cons 'buttercup ()) -> (buttercup)

LENGTH -> returneaza lungimea listei
   (length '(daisy buttercup)) -> 2

NTHCDR -> face cdr de mai multe ori
    (cdr (cdr '(pine fir oak maple)))
        => (oak maple)

(nthcdr 2 '(pine fir oak maple))
     => (oak maple)

SETQ -> seteaza o variabila catre o lista
    (setq animals '(giraffe antelope tiger lion)) : animals -> (giraffe antelope tiger lion)

SETCAR -> seteaza primul element la o valoare data
    (setcar animals 'hippopotamus) : animals -> (hippopotamus antelope tiger lion)

SETCDR -> seteaza restul listei
  (setq domesticated-animals '(horse cow sheep goat)) : domesticated-animals -> (horse cow sheep goat)
  (setcdr domesticated-animals '(cat dog)) : domesticated-animals -> (horse cat dog)

HOW LISTS ARE IMPLEMENTED
  (rose violet buttercup)
    ___ ___      ___ ___      ___ ___ 
   |___|___|--> |___|___|--> |___|___|--> nil
     |            |            |      
     |            |            |
      --> rose     --> violet   --> buttercup

  (setq bouquet '(rose violet buttercup))

  bouquet
     |
     |     ___ ___      ___ ___      ___ ___ 
      --> |___|___|--> |___|___|--> |___|___|--> nil
            |            |            |      
            |            |            |
             --> rose     --> violet   --> buttercup

  (setq flowers (cdr bouquet))

   produces this:

    bouquet        flowers
      |              |        
      |     ___ ___  |     ___ ___      ___ ___ 
       --> |   |   |  --> |   |   |    |   |   |
           |___|___|----> |___|___|--> |___|___|--> nil
             |              |            |      
             |              |            |
              --> rose       --> violet   --> buttercup


  (setq bouquet (cons 'lilly bouquet))
  
   produces:

    bouquet                       flowers        
      |                             |                 
      |     ___ ___        ___ ___  |     ___ ___       ___ ___        
       --> |   |   |      |   |   |  --> |   |   |     |   |   |       
           |___|___|----> |___|___|----> |___|___|---->|___|___|--> nil
             |              |              |             |             
             |              |              |             |             
              --> lilly      --> rose       --> violet    --> buttercup
    
    (eq (cdr (cdr bouquet)) flowers) -> true
